\begin{align}
  \label{eq:optimization_objective}
  \mathcal{L}(\theta) &= \mathbb{E}_{x \sim \mathcal{D}} \left[ -\log p_\theta(x) \right] + \lambda \mathcal{R}(\theta) \\
  &= \frac{1}{N} \sum_{i=1}^{N} \ell(f_\theta(x_i), y_i) + \frac{\lambda}{2} \|\theta\|_2^2.
\end{align}
