\section{Principes Éditoriaux}

\begin{frame}{Rupture volontaire avec le rendu Beamer traditionnel}
  \framesubtitle{Privilégier la hiérarchie visuelle, l'air typographique et la lisibilité}

  \begin{columns}[T, totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \begin{alertcard}{\faTimesCircle\enspace À proscrire}
        \begin{itemize}
          \item Bandeaux pleins sombres et bordures lourdes
          \item Ombres portées datées des années 2000
          \item Paragraphes denses ($> 50$ mots par vue)
          \item Symboles de navigation superflus en pied de page
        \end{itemize}
      \end{alertcard}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{slidecard}{\faCheckCircle\enspace Approche éditoriale 16:9}
        \begin{itemize}
          \item Toile blanche pure (ou ardoise en \code{theme=dark})
          \item Titre-action affirmé et surtitre discret
          \item Palette unifiée avec \code{ClasseGuillaume}~\autocite{bringhurst2004elements}
          \item Barre de progression fine et pagination sobre
        \end{itemize}
      \end{slidecard}
    \end{column}
  \end{columns}

  \takeaway{\textbf{Idée directrice :} chaque diapositive transmet un seul message clair, vérifié en moins de \qty{5}{\second} de lecture.}
\end{frame}
